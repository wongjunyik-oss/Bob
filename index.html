import pygame import pymunk import pymunk.pygame_util import random # --- CONFIGURATION --- WIDTH, HEIGHT = 800, 600 FPS = 60 class Ragdoll: def __init__(self, space, pos, color, is_player=True): self.space = space self.color = color self.is_player = is_player # 1. Create Core Body (Torso) self.torso = self._add_box(pos, (20, 40), 1.0) # 2. Create Head self.head = self._add_circle((pos[0], pos[1]-35), 10, 0.5) self._connect(self.torso, self.head, (0, -20), (0, 10)) # 3. Create Arms (Limbs) self.r_arm = self._add_box((pos[0]+25, pos[1]-10), (25, 10), 0.5) self._connect(self.torso, self.r_arm, (10, -15), (-12, 0)) self.l_arm = self._add_box((pos[0]-25, pos[1]-10), (25, 10), 0.5) self._connect(self.torso, self.l_arm, (-10, -15), (12, 0)) def _add_box(self, pos, size, mass): body = pymunk.Body(mass, pymunk.moment_for_box(mass, size)) body.position = pos shape = pymunk.Poly.create_box(body, size) shape.friction, shape.color = 0.5, self.color self.space.add(body, shape) return body def _add_circle(self, pos, radius, mass): body = pymunk.Body(mass, pymunk.moment_for_circle(mass, 0, radius)) body.position = pos shape = pymunk.Circle(body, radius) shape.friction, shape.color = 0.5, self.color self.space.add(body, shape) return body def _connect(self, b1, b2, a1, a2): self.space.add(pymunk.PivotJoint(b1, b2, b1.local_to_world(a1))) self.space.add(pymunk.RotaryLimitJoint(b1, b2, -0.6, 0.6)) def move(self, direction): # Applies force to the torso for horizontal movement self.torso.apply_impulse_at_local_point((direction * 400, -150)) def punch(self): # Flails an arm forward arm = random.choice([self.r_arm, self.l_arm]) force = 600 if self.is_player else -600 arm.apply_impulse_at_local_point((force, -100)) def run_game(): pygame.init() screen = pygame.display.set_mode((WIDTH, HEIGHT)) clock = pygame.time.Clock() draw_options = pymunk.pygame_util.DrawOptions(screen) space = pymunk.Space() space.gravity = (0, 900) # Static Floor floor = pymunk.Segment(space.static_body, (0, HEIGHT-50), (WIDTH, HEIGHT-50), 5) floor.friction = 1.0 space.add(floor) player = Ragdoll(space, (200, 400), (0, 0, 255), True) ai = Ragdoll(space, (600, 400), (255, 0, 0), False) ai_intelligence = 1.0 # Scales up over time while True: for event in pygame.event.get(): if event.type == pygame.QUIT: return if event.type == pygame.KEYDOWN and event.key == pygame.K_SPACE: player.punch() # --- Dual Controls (WASD & Arrows) --- keys = pygame.key.get_pressed() if keys[pygame.K_a] or keys[pygame.K_LEFT]: player.move(-1) if keys[pygame.K_d] or keys[pygame.K_RIGHT]: player.move(1) # --- "Smarter" AI Logic --- dist_x = player.torso.position.x - ai.torso.position.x # AI moves toward player with increasing speed if abs(dist_x) > 60: dir = 1 if dist_x > 0 else -1 ai.move(dir * (0.6 + ai_intelligence * 0.1)) # AI punches more frequently as difficulty scales if abs(dist_x) < 90 and random.random() < (0.02 * ai_intelligence): ai.punch() ai_intelligence += 0.002 # Gradually gets "smarter"/harder screen.fill((240, 240, 240)) space.debug_draw(draw_options) space.step(1/FPS) pygame.display.flip() clock.tick(FPS) if __name__ == "__main__": run_game()
